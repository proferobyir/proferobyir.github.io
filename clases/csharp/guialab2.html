1. Saludador Múltiple
Solicita al usuario que ingrese nombres de estudiantes uno a uno y los guarde en una List<string> hasta que escriba "fin". Al terminar, recorre la lista con un bucle foreach y saluda a cada estudiante.
C# 
using System;
using System.Collections.Generic;

class Program
{
    static void Main()
    {
        List<string> estudiantes = new List<string>();
        string nombre;

        Console.WriteLine("Ingrese los nombres de los estudiantes (escriba 'fin' para terminar):");
        while (true)
        {
            Console.Write("Nombre: ");
            nombre = Console.ReadLine();
            if (nombre.ToLower() == "fin") break;
            estudiantes.Add(nombre);
        }

        Console.WriteLine("
--- Saludos ---");
        foreach (string est in estudiantes)
        {
            Console.WriteLine($"¡Hola, {est}! Bienvenido a la clase.");
        }
    }
}

4. Facturador con Menú de Impuestos
Pide el precio base de un producto. Presenta un menú mediante switch para seleccionar el tipo de IVA (16%, 8%, 0%). Repite el menú si selecciona una opción inválida usando un bucle do-while.
C# 
using System;

class Program
{
    static void Main()
    {
        Console.Write("Ingrese el precio base del producto: ");
        double precioBase = double.Parse(Console.ReadLine());
        int opcion;
        double precioFinal = 0;
        bool opcionValida = true;

        do
        {
            Console.WriteLine("
--- Seleccione el IVA ---");
            Console.WriteLine("1. General (16%)");
            Console.WriteLine("2. Reducido (8%)");
            Console.WriteLine("3. Exento (0%)");
            Console.Write("Opción: ");
            opcion = int.Parse(Console.ReadLine());

            switch (opcion)
            {
                case 1:
                    precioFinal = precioBase * 1.16;
                    opcionValida = true;
                    break;
                case 2:
                    precioFinal = precioBase * 1.08;
                    opcionValida = true;
                    break;
                case 3:
                    precioFinal = precioBase;
                    opcionValida = true;
                    break;
                default:
                    Console.WriteLine("Opción inválida. Intente de nuevo.");
                    opcionValida = false;
                    break;
            }
        } while (!opcionValida);

        Console.WriteLine($"El precio final a pagar es: ${precioFinal:F2}");
    }
}
5. Contador de Vocales y Consonantes
Permite ingresar caracteres en una List<char> hasta presionar la tecla '0'. Usa un foreach con un switch para contar vocales y consonantes.
C# 
using System;
using System.Collections.Generic;

class Program
{
    static void Main()
    {
        List<char> caracteres = new List<char>();
        char c;

        Console.WriteLine("Ingrese letras una a una (ingrese '0' para finalizar):");
        while (true)
        {
            c = Console.ReadKey().KeyChar;
            Console.WriteLine();
            if (c == '0') break;
            caracteres.Add(char.ToLower(c));
        }

        int vocales = 0, consonantes = 0;
        foreach (char item in caracteres)
        {
            if (char.IsLetter(item))
            {
                switch (item)
                {
                    case 'a': case 'e': case 'i': case 'o': case 'u':
                        vocales++;
                        break;
                    default:
                        consonantes++;
                        break;
                }
            }
        }
        Console.WriteLine($"
Vocales: {vocales} | Consonantes: {consonantes}");
    }
}

7. Clasificador de Números
Pide al usuario N números. Almacénalos en una lista y recorriéndola, clasifícalos mostrando si cada uno es positivo, negativo o cero.
C# 
using System;
using System.Collections.Generic;

class Program
{
    static void Main()
    {
        List<int> numeros = new List<int>();
        Console.Write("Cantidad de números a ingresar: ");
        int cantidad = int.Parse(Console.ReadLine());

        for (int i = 0; i < cantidad; i++)
        {
            Console.Write($"Número {i + 1}: ");
            numeros.Add(int.Parse(Console.ReadLine()));
        }

        Console.WriteLine("
--- Resultados ---");
        foreach (int num in numeros)
        {
            int signo = Math.Sign(num); // Retorna 1 (Positivo), -1 (Negativo) o 0
            switch (signo)
            {
                case 1:
                    Console.WriteLine($"El número {num} es POSITIVO.");
                    break;
                case -1:
                    Console.WriteLine($"El número {num} es NEGATIVO.");
                    break;
                case 0:
                    Console.WriteLine($"El número es CERO.");
                    break;
            }
        }
    }
}

9. Búsqueda del Número Mayor
Pide N números y guárdalos en una lista. Usa un bucle foreach para determinar el número mayor ingresado y cuántas veces se repite.
C# 
using System;
using System.Collections.Generic;

class Program
{
    static void Main()
    {
        List<int> numeros = new List<int>();
        Console.Write("¿Cuántos números ingresará?: ");
        int total = int.Parse(Console.ReadLine());

        for (int i = 0; i < total; i++)
        {
            Console.Write($"Elemento {i + 1}: ");
            numeros.Add(int.Parse(Console.ReadLine()));
        }

        int mayor = int.MinValue;
        foreach (int n in numeros)
        {
            if (n > mayor) mayor = n;
        }

        int repeticiones = 0;
        foreach (int n in numeros)
        {
            if (n == mayor) repeticiones++;
        }

        Console.WriteLine($"$"
El número mayor es: {mayor} (se repite {repeticiones} veces)");
    }
}
10. Registro de Notas y Promedio
Permite cargar calificaciones (0 a 10) en una List<float> hasta ingresar -1. Muestra la escala equivalente mediante switch y calcula el promedio.
C# 
using System;
using System.Collections.Generic;

class Program
{
    static void Main()
    {
        List<float> notas = new List<float>();
        float nota = 0;

        Console.WriteLine("Ingrese notas de 0 a 10 (-1 para finalizar):");
        while (true)
        {
            Console.Write("Nota: ");
            nota = float.Parse(Console.ReadLine());
            if (nota == -1) break;
            if (nota >= 0 && nota <= 10) notas.Add(nota);
            else Console.WriteLine("Nota inválida.");
        }

        float suma = 0;
        Console.WriteLine("
--- Detalle de Notas ---");
        foreach (float n in notas)
        {
            suma += n;
            int notaEntera = (int)Math.Floor(n);
            string escala = "";
            switch (notaEntera)
            {
                case 10: case 9: escala = "Sobresaliente"; break;
                case 8: case 7: escala = "Notable"; break;
                case 6: case 5: escala = "Aprobado"; break;
                default: escala = "Reprobado"; break;
            }
            Console.WriteLine($"Nota: {n} -> {escala}");
        }

        if (notas.Count > 0)
            Console.WriteLine($"$"
Promedio General: {suma / notas.Count:F2}");
    }
}

16. Censo por Categoría de Edad
Solicita el año de nacimiento de N personas. Calcula sus edades y las clasifica en Niños, Adolescentes, Adultos y Adultos Mayores.
C# 
using System;
using System.Collections.Generic;

class Program
{
    static void Main()
    {
        List<int> edades = new List<int>();
        int anioActual = DateTime.Now.Year;

        Console.Write("¿Cuántas personas censará?: ");
        int n = int.Parse(Console.ReadLine());

        for (int i = 0; i < n; i++)
        {
            Console.Write($"Año de nacimiento de persona {i + 1}: ");
            int anioNac = int.Parse(Console.ReadLine());
            edades.Add(anioActual - anioNac);
        }

        int ninos = 0, adolescentes = 0, adultos = 0, mayores = 0;

        foreach (int edad in edades)
        {
            if (edad <= 12) ninos++;
            else if (edad <= 17) adolescentes++;
            else if (edad <= 59) adultos++;
            else mayores++;
        }

        Console.WriteLine("
--- Resultados del Censo ---");
        Console.WriteLine($"Niños: {ninos}
Adolescentes: {adolescentes}
Adultos: {adultos}
Adultos Mayores: {mayores}");
    }
}

class Program

    {

    }

20. Calculadora Grupal de IMC
Registra nombres, pesos y alturas en listas paralelas. Muestra un reporte final usando un ciclo for indicando el estado de cada paciente.
C# 
using System;
using System.Collections.Generic;

class Program
{
    static void Main()
    {
        List<string> nombres = new List<string>();
        List<double> imcs = new List<double>();

        Console.Write("¿Cuántos pacientes registrará?: ");
        int n = int.Parse(Console.ReadLine());

        for (int i = 0; i < n; i++)
        {
            Console.Write($"$"
Nombre paciente {i + 1}: ");
            nombres.Add(Console.ReadLine());
            
            Console.Write("Peso (kg): ");
            double peso = double.Parse(Console.ReadLine());
            
            Console.Write("Altura (m): ");
            double altura = double.Parse(Console.ReadLine());

            double imc = peso / (altura * altura);
            imcs.Add(imc);
        }

        Console.WriteLine("
--- REPORTE FINAL IMC ---");
        for (int i = 0; i < nombres.Count; i++)
        {
            string diagnostico = "";
            if (imcs[i] < 18.5) diagnostico = "Bajo peso";
            else if (imcs[i] <= 24.9) diagnostico = "Peso normal";
            else diagnostico = "Sobrepeso";

            Console.WriteLine($"$"{nombres[i]} -> IMC: {imcs[i]:F2} ({diagnostico})");
        }
    }
}
