Paso 2: Lectura de Datos y Carga del ComboBox

Conecta a MySQL, mapea los registros a una List<Departamento> y enlaza el control. Nota: comboBoxDepartamentos debe existir en el formulario. La cadena de conexión se centraliza en ConexionDB.

// Datos/ConexionDB.cs
namespace LaboratorioADO.NET.Datos
{
    public static class ConexionDB
    {
        public const string CadenaConexion =
            "Server=localhost;Database=tu_bd;Uid=root;Pwd=tu_pass;SslMode=none;";
    }
}

// Datos/DepartamentoDatos.cs
using System;
using System.Collections.Generic;
using System.Windows.Forms;
using MySql.Data.MySqlClient;
using LaboratorioADO.NET.Modelos;

namespace LaboratorioADO.NET.Datos
{
    public class DepartamentoDatos
    {
        public List<Departamento> ObtenerTodos()
        {
            List<Departamento> lista = new List<Departamento>();
            string query = "SELECT id, nombre FROM departamentos ORDER BY nombre ASC;";

            try
            {
                using (MySqlConnection conexion = new MySqlConnection(ConexionDB.CadenaConexion))
                {
                    conexion.Open();
                    using (MySqlCommand comando = new MySqlCommand(query, conexion))
                    using (MySqlDataReader lector = comando.ExecuteReader())
                    {
                        while (lector.Read())
                        {
                            Departamento depto = new Departamento();
                            depto.Id = lector.GetInt32("id");
                            depto.Nombre = lector.GetString("nombre");
                            lista.Add(depto);
                        }
                    }
                }
            }
            catch (MySqlException ex)
            {
                MessageBox.Show("Error de MySQL: " + ex.Message, "Error",
                    MessageBoxButtons.OK, MessageBoxIcon.Error);
            }

            return lista;
        }
    }
}

En el formulario, enlaza la lista al ComboBox:

private void CargarDepartamentosEnComboBox()
{
    DepartamentoDatos datos = new DepartamentoDatos();
    List<Departamento> lista = datos.ObtenerTodos();

    // Opción por defecto al inicio
    Departamento opcionDefecto = new Departamento();
    opcionDefecto.Id = 0;
    opcionDefecto.Nombre = "-- Seleccione un Departamento --";
    lista.Insert(0, opcionDefecto);

    // Enlace con ComboBox
    comboBoxDepartamentos.DataSource = lista;
    comboBoxDepartamentos.DisplayMember = "Nombre";
    comboBoxDepartamentos.ValueMember = "Id";
    comboBoxDepartamentos.SelectedIndex = 0;
}

Paso 3: Inserción Segura mediante Parámetros

Guarda una instancia de Usuario usando parámetros SQL:

// Datos/UsuarioDatos.cs
using System.Windows.Forms;
using MySql.Data.MySqlClient;
using LaboratorioADO.NET.Modelos;

namespace LaboratorioADO.NET.Datos
{
    public class UsuarioDatos
    {
        public bool Guardar(Usuario nuevoUsuario)
        {
            string query = "INSERT INTO usuarios (nickname, password_hash, correo, id_departamento) " +
                           "VALUES (@nickname, @password, @correo, @idDepto);";

            try
            {
                using (MySqlConnection conexion = new MySqlConnection(ConexionDB.CadenaConexion))
                using (MySqlCommand comando = new MySqlCommand(query, conexion))
                {
                    comando.Parameters.AddWithValue("@nickname", nuevoUsuario.Nickname);
                    comando.Parameters.AddWithValue("@password", nuevoUsuario.PasswordHash);
                    comando.Parameters.AddWithValue("@correo", nuevoUsuario.Correo);
                    comando.Parameters.AddWithValue("@idDepto", nuevoUsuario.IdDepartamento);

                    conexion.Open();
                    int filasAfectadas = comando.ExecuteNonQuery();
                    return filasAfectadas > 0;
                }
            }
            catch (MySqlException ex)
            {
                MessageBox.Show("Error al guardar: " + ex.Message, "Error",
                    MessageBoxButtons.OK, MessageBoxIcon.Error);
                return false;
            }
        }
    }
}

Paso 4: Formulario Completo de Ejemplo

Formulario construido por código (sin diseñador) para copiar y pegar. Controles: comboBoxDepartamentos, txtNickname, txtPassword, txtCorreo, btnGuardar y lblEstado.

// Formularios/FrmUsuarios.cs
using System;
using System.Collections.Generic;
using System.Drawing;
using System.Windows.Forms;
using LaboratorioADO.NET.Datos;
using LaboratorioADO.NET.Modelos;

namespace LaboratorioADO.NET.Formularios
{
    public class FrmUsuarios : Form
    {
        // Controles
        private ComboBox comboBoxDepartamentos;
        private TextBox txtNickname;
        private TextBox txtPassword;
        private TextBox txtCorreo;
        private Button btnGuardar;
        private Label lblEstado;

        // Dependencias
        private readonly DepartamentoDatos departamentoDatos = new DepartamentoDatos();
        private readonly UsuarioDatos usuarioDatos = new UsuarioDatos();

        public FrmUsuarios()
        {
            InicializarControles();
            CargarDepartamentosEnComboBox();
        }

        private void InicializarControles()
        {
            this.Text = "Registro de Usuarios - Laboratorio ADO.NET";
            this.Width = 460;
            this.Height = 340;
            this.StartPosition = FormStartPosition.CenterScreen;
            this.FormBorderStyle = FormBorderStyle.FixedDialog;
            this.MaximizeBox = false;

            Label lblDepto = new Label();
            lblDepto.Text = "Departamento:";
            lblDepto.SetBounds(20, 20, 120, 20);

            comboBoxDepartamentos = new ComboBox();
            comboBoxDepartamentos.SetBounds(150, 18, 270, 24);
            comboBoxDepartamentos.DropDownStyle = ComboBoxStyle.DropDownList;
            comboBoxDepartamentos.SelectedIndexChanged += comboBoxDepartamentos_SelectedIndexChanged;

            Label lblNick = new Label();
            lblNick.Text = "Nickname:";
            lblNick.SetBounds(20, 60, 120, 20);

            txtNickname = new TextBox();
            txtNickname.SetBounds(150, 58, 270, 22);

            Label lblPass = new Label();
            lblPass.Text = "Password:";
            lblPass.SetBounds(20, 100, 120, 20);

            txtPassword = new TextBox();
            txtPassword.SetBounds(150, 98, 270, 22);
            txtPassword.UseSystemPasswordChar = true;

            Label lblCorreo = new Label();
            lblCorreo.Text = "Correo:";
            lblCorreo.SetBounds(20, 140, 120, 20);

            txtCorreo = new TextBox();
            txtCorreo.SetBounds(150, 138, 270, 22);

            btnGuardar = new Button();
            btnGuardar.Text = "Guardar Usuario";
            btnGuardar.SetBounds(150, 180, 270, 32);
            btnGuardar.Click += btnGuardar_Click;

            lblEstado = new Label();
            lblEstado.SetBounds(20, 230, 400, 50);
            lblEstado.ForeColor = Color.DarkSlateGray;
            lblEstado.Text = "Seleccione un departamento.";

            this.Controls.Add(lblDepto);
            this.Controls.Add(comboBoxDepartamentos);
            this.Controls.Add(lblNick);
            this.Controls.Add(txtNickname);
            this.Controls.Add(lblPass);
            this.Controls.Add(txtPassword);
            this.Controls.Add(lblCorreo);
            this.Controls.Add(txtCorreo);
            this.Controls.Add(btnGuardar);
            this.Controls.Add(lblEstado);
        }

        private void btnGuardar_Click(object sender, EventArgs e)
        {
            Departamento depto = comboBoxDepartamentos.SelectedItem as Departamento;
            if (depto == null || depto.Id == 0)
            {
                MessageBox.Show("Debe seleccionar un departamento válido.", "Validación",
                    MessageBoxButtons.OK, MessageBoxIcon.Warning);
                return;
            }

            if (string.IsNullOrWhiteSpace(txtNickname.Text) ||
                string.IsNullOrWhiteSpace(txtPassword.Text) ||
                string.IsNullOrWhiteSpace(txtCorreo.Text))
            {
                MessageBox.Show("Complete todos los campos.", "Validación",
                    MessageBoxButtons.OK, MessageBoxIcon.Warning);
                return;
            }

            Usuario u = new Usuario();
            u.Nickname = txtNickname.Text.Trim();
            u.PasswordHash = txtPassword.Text; // En producción: hashear (SHA256/BCrypt)
            u.Correo = txtCorreo.Text.Trim();
            u.IdDepartamento = depto.Id;

            bool ok = usuarioDatos.Guardar(u);

            if (ok)
            {
                MessageBox.Show("Usuario guardado correctamente.", "Éxito",
                    MessageBoxButtons.OK, MessageBoxIcon.Information);
                txtNickname.Clear();
                txtPassword.Clear();
                txtCorreo.Clear();
                comboBoxDepartamentos.SelectedIndex = 0;
            }
        }
    }
}

Ejercicio Práctico: Evento SelectedIndexChanged

Captura la selección activa haciendo casting directo a Departamento (C# 5):

private void comboBoxDepartamentos_SelectedIndexChanged(object sender, EventArgs e)
{
    Departamento seleccionado = comboBoxDepartamentos.SelectedItem as Departamento;

    if (seleccionado != null && seleccionado.Id > 0)
    {
        lblEstado.Text = "Departamento: " + seleccionado.Nombre +
                         " | ID: " + seleccionado.Id.ToString();
    }
    else
    {
        lblEstado.Text = "Por favor, seleccione un departamento válido.";
    }
}
