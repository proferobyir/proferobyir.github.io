Paso 2: Extracción, Mapeo a Lista y Carga del ComboBox

Implementa la conexión usando MySqlConnection, ejecuta la consulta con MySqlCommand, lee fila por fila mediante MySqlDataReader e integra la opción por defecto ("-- Seleccione --"):

using System;
using System.Collections.Generic;
using System.Windows.Forms;
using MySql.Data.MySqlClient;
​private void CargarDepartamentosEnComboBox()
{
string connectionString = "Server=localhost;Database=tu_bd;Uid=root;Pwd=tu_pass;SslMode=none;";
List listaDepartamentos = new List();
string query = "SELECT id, nombre FROM departamentos ORDER BY nombre ASC;";
​// 1. OBJETO QUE SE CONECTA
using (MySqlConnection conexion = new MySqlConnection(connectionString))
{
// 2. OBJETO QUE EJECUTA LA CONSULTA
using (MySqlCommand comando = new MySqlCommand(query, conexion))
{
try
{
conexion.Open();
​// 3. OBJETO QUE LEE LA DATA
using (MySqlDataReader lector = comando.ExecuteReader())
{
while (lector.Read())
{
// Mapeo directo a objeto
Departamento depto = new Departamento
{
Id = lector.GetInt32("id"),
Nombre = lector.GetString("nombre")
};
listaDepartamentos.Add(depto);
}
}
}
catch (MySqlException ex)
{

}

comboBoxDepartamentos.DataSource = listaDepartamentos;

Paso 3: Inserción Segura de Datos utilizando Parámetros

Utiliza parámetros nombrados en MySqlCommand para insertar un nuevo objeto Usuario de forma segura (previniendo SQL Injection):

public bool GuardarUsuario(Usuario nuevoUsuario)
{
string connectionString = "Server=localhost;Database=tu_bd;Uid=root;Pwd=tu_pass;SslMode=none;";
string query = "INSERT INTO usuarios (nickname, password_hash, correo, id_departamento) " +
"VALUES (@nickname, @password, @correo, @idDepto);";
​using (MySqlConnection conexion = new MySqlConnection(connectionString))
{
using (MySqlCommand comando = new MySqlCommand(query, conexion))
{
// Parametrización segura de variables
comando.Parameters.AddWithValue("@nickname", nuevoUsuario.Nickname);
comando.Parameters.AddWithValue("@password", nuevoUsuario.PasswordHash);
comando.Parameters.AddWithValue("@correo", nuevoUsuario.Correo);
comando.Parameters.AddWithValue("@idDepto", nuevoUsuario.IdDepartamento);
​try
{
conexion.Open();
int filasAfectadas = comando.ExecuteNonQuery();
return filasAfectadas > 0;
}
catch (MySqlException ex)
{
MessageBox.Show("Error al guardar: " + ex.Message);
return false;
}
}
}
}
