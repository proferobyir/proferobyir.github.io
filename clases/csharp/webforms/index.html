ASP.NET Web Forms + MySQL con Clases Públicas, List y Dictionary

Guía paso a paso para construir un sistema web con inicio de sesión, catálogo dinámico de productos, registro de productos y registro de usuarios usando Bootstrap.

1. Base de Datos 2. Web.config 3. Clases (POO) 4. Login y Diccionario 5. Registro Usuarios 6. Dashboard y List<T> 7. Registro Productos
Paso 1

Crear la Base de Datos y Tablas en MySQL

Antes de abrir Visual Studio, crea la base de datos tienda_webforms en MySQL con dos tablas:

  • usuarios: almacena las credenciales para iniciar sesión y el rol (admin o usuario).
  • productos: guarda los artículos que se cargarán dinámicamente en el catálogo.
base_datos.sql 
CREATE DATABASE IF NOT EXISTS tienda_webforms;
USE tienda_webforms;

CREATE TABLE usuarios (
    id INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(100) NOT NULL,
    usuario VARCHAR(60) NOT NULL UNIQUE,
    clave VARCHAR(100) NOT NULL,
    rol VARCHAR(30) DEFAULT 'usuario'
);

CREATE TABLE productos (
    id INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(120) NOT NULL,
    descripcion VARCHAR(255),
    precio DECIMAL(10,2) NOT NULL,
    stock INT NOT NULL DEFAULT 0,
    imagen_url VARCHAR(255)
);

-- Usuario inicial para pruebas en clase
INSERT INTO usuarios (nombre, usuario, clave, rol)

Paso 2

Configurar el Proyecto Web Forms (.NET 4.5) y Web.config

En Visual Studio, crea un proyecto Aplicación web ASP.NET (.NET Framework 4.5) con plantilla Web Forms (o vacía y agregas los Web Forms).

  • Instala el paquete NuGet MySql.Data compatible con .NET Framework 4.5 (o agrega la referencia a MySql.Data.dll).
  • Agrega la cadena de conexión en Web.config dentro de <configuration> para centralizar el acceso a MySQL.
Web.config 
<configuration>
  <connectionStrings>
    <add name="MySqlConn"
         connectionString="Server=localhost;Database=tienda_webforms;Uid=root;Pwd=;"
         providerName="MySql.Data.MySqlClient" />
  </connectionStrings>

  <system.web>
    <compilation debug="true" targetFramework="4.5" />
    <httpRuntime targetFramework="4.5" />
  </system.web>
</configuration>

Login.aspx.cs 
using System;
using System.Collections.Generic;
using System.Configuration;
using MySql.Data.MySqlClient;

namespace TiendaWeb
{
    public partial class Login : System.Web.UI.Page
    {
        protected void btnIngresar_Click(object sender, EventArgs e)
        {
            Dictionary<string, string> mensajes = new Dictionary<string, string>
            {
                { "vacio", "Por favor, completa usuario y contraseña." },
                { "invalido", "Usuario o contraseña incorrectos." }
            };

            string user = txtUsuario.Text.Trim();
            string pass = txtClave.Text.Trim();

            if (string.IsNullOrEmpty(user) || string.IsNullOrEmpty(pass))
            {
                MostrarError(mensajes["vacio"]);
                return;
            }

            string connStr = ConfigurationManager.ConnectionStrings["MySqlConn"].ConnectionString;

            using (MySqlConnection conn = new MySqlConnection(connStr))
            {
                conn.Open();
                string sql = "SELECT id, nombre, usuario, rol FROM usuarios WHERE usuario = @usuario AND clave = @clave LIMIT 1";

                using (MySqlCommand cmd = new MySqlCommand(sql, conn))
                {
                    cmd.Parameters.AddWithValue("@usuario", user);
                    cmd.Parameters.AddWithValue("@clave", pass);

                    using (MySqlDataReader reader = cmd.ExecuteReader())
                    {
                        if (reader.Read())
                        {
                            Usuario usuarioActual = new Usuario
                            {
                                Id = Convert.ToInt32(reader["id"]),
                                Nombre = reader["nombre"].ToString(),
                                UsuarioLogin = reader["usuario"].ToString(),
                                Rol = reader["rol"].ToString()
                            };

                            Session["UsuarioActual"] = usuarioActual;
                            Response.Redirect("Dashboard.aspx");
                        }
                        else
                        {
                            MostrarError(mensajes["invalido"]);
                        }
                    }
                }
            }
        }

        private void MostrarError(string texto)
        {
            lblError.Text = texto;
            pnlError.Visible = true;
        }
    }
}

using System;
using System.Collections.Generic;
using System.Configuration;
using MySql.Data.MySqlClient;

namespace TiendaWeb
{
    public partial class RegistroUsuario : System.Web.UI.Page
    {
        protected void Page_Load(object sender, EventArgs e)
        {
            if (!IsPostBack)
            {
                Dictionary<string, string> roles = new Dictionary<string, string>
                {
                    { "usuario", "Usuario Estándar" },
                    { "admin", "Administrador" }
                };

                ddlRol.DataSource = roles;
                ddlRol.DataTextField = "Value";
                ddlRol.DataValueField = "Key";
                ddlRol.DataBind();
            }
        }

        protected void btnGuardar_Click(object sender, EventArgs e)
        {
            if (string.IsNullOrWhiteSpace(txtNombre.Text) ||
                string.IsNullOrWhiteSpace(txtUsuario.Text) ||
                string.IsNullOrWhiteSpace(txtClave.Text))
            {
                lblMensaje.Text = "<div class='alert alert-warning'>Todos los campos son obligatorios.</div>";
                return;
            }

            string connStr = ConfigurationManager.ConnectionStrings["MySqlConn"].ConnectionString;

            try
            {
                using (MySqlConnection conn = new MySqlConnection(connStr))
                {
                    conn.Open();
                    string sql = "INSERT INTO usuarios (nombre, usuario, clave, rol) VALUES (@nombre, @usuario, @clave, @rol)";

                    using (MySqlCommand cmd = new MySqlCommand(sql, conn))
                    {
                        cmd.Parameters.AddWithValue("@nombre", txtNombre.Text.Trim());
                        cmd.Parameters.AddWithValue("@usuario", txtUsuario.Text.Trim());
                        cmd.Parameters.AddWithValue("@clave", txtClave.Text.Trim());
                        cmd.Parameters.AddWithValue("@rol", ddlRol.SelectedValue);
                        cmd.ExecuteNonQuery();
                    }
                }

                lblMensaje.Text = "<div class='alert alert-success'>Usuario registrado correctamente.</div>";
                txtNombre.Text = "";
                txtUsuario.Text = "";
                txtClave.Text = "";
            }
            catch (MySqlException ex)
            {
                lblMensaje.Text = "<div class='alert alert-danger'>Error al registrar: " + ex.Message + "</div>";
            }
        }
    }
}

Dashboard.aspx.cs 
using System;
using System.Collections.Generic;
using System.Configuration;
using MySql.Data.MySqlClient;

namespace TiendaWeb
{
    public partial class Dashboard : System.Web.UI.Page
    {
        protected void Page_Load(object sender, EventArgs e)
        {
            if (Session["UsuarioActual"] == null)
            {
                Response.Redirect("Login.aspx");
                return;
            }

            if (!IsPostBack)
            {
                Usuario user = (Usuario)Session["UsuarioActual"];
                lblBienvenida.Text = user.Nombre;

                Dictionary<string, string> etiquetasRol = new Dictionary<string, string>
                {
                    { "admin", "Administrador" },
                    { "usuario", "Usuario Estándar" }
                };

                lblRol.Text = etiquetasRol.ContainsKey(user.Rol) ? etiquetasRol[user.Rol] : user.Rol;

                CargarCatalogo();
            }
        }

        private void CargarCatalogo()
        {
            string connStr = ConfigurationManager.ConnectionStrings["MySqlConn"].ConnectionString;
            List<Producto> listaProductos = new List<Producto>();

            using (MySqlConnection conn = new MySqlConnection(connStr))
            {
                conn.Open();
                string sql = "SELECT id, nombre, descripcion, precio, stock, imagen_url FROM productos ORDER BY id DESC";

                using (MySqlCommand cmd = new MySqlCommand(sql, conn))
                using (MySqlDataReader reader = cmd.ExecuteReader())
                {
                    while (reader.Read())
                    {
                        Producto p = new Producto
                        {
                            Id = Convert.ToInt32(reader["id"]),
                            Nombre = reader["nombre"].ToString(),
                            Descripcion = reader["descripcion"].ToString(),
                            Precio = Convert.ToDecimal(reader["precio"]),
                            Stock = Convert.ToInt32(reader["stock"]),
                            ImagenUrl = string.IsNullOrWhiteSpace(reader["imagen_url"].ToString())
                                        ? "https://via.placeholder.com/400x220?text=Sin+Imagen"
                                        : reader["imagen_url"].ToString()
                        };

                        listaProductos.Add(p);
                    }
                }
            }

            rptProductos.DataSource = listaProductos;
            rptProductos.DataBind();
        }

        protected void btnSalir_Click(object sender, EventArgs e)
        {
            Session.Clear();
            Session.Abandon();
            Response.Redirect("Login.aspx");
        }
    }
}

RegistroProducto.aspx.cs 
using System;
using System.Configuration;
using MySql.Data.MySqlClient;

namespace TiendaWeb
{
    public partial class RegistroProducto : System.Web.UI.Page
    {
        protected void Page_Load(object sender, EventArgs e)
        {
            if (Session["UsuarioActual"] == null)
            {
                Response.Redirect("Login.aspx");
            }
        }

        protected void btnRegistrar_Click(object sender, EventArgs e)
        {
            decimal precio;
            int stock;

            if (string.IsNullOrWhiteSpace(txtNombre.Text) ||
                !decimal.TryParse(txtPrecio.Text.Trim(), out precio) ||
                !int.TryParse(txtStock.Text.Trim(), out stock))
            {
                lblEstado.Text = "<div class='alert alert-warning'>Verifica el nombre, el precio y el stock numérico.</div>";
                return;
            }

            Producto nuevo = new Producto
            {
                Nombre = txtNombre.Text.Trim(),
                Descripcion = txtDescripcion.Text.Trim(),
                Precio = precio,
                Stock = stock,
                ImagenUrl = txtImagen.Text.Trim()
            };

            string connStr = ConfigurationManager.ConnectionStrings["MySqlConn"].ConnectionString;

            using (MySqlConnection conn = new MySqlConnection(connStr))
            {
                conn.Open();
                string sql = @"INSERT INTO productos (nombre, descripcion, precio, stock, imagen_url)
                               VALUES (@nombre, @descripcion, @precio, @stock, @imagen)";

                using (MySqlCommand cmd = new MySqlCommand(sql, conn))
                {
                    cmd.Parameters.AddWithValue("@nombre", nuevo.Nombre);
                    cmd.Parameters.AddWithValue("@descripcion", nuevo.Descripcion);
                    cmd.Parameters.AddWithValue("@precio", nuevo.Precio);
                    cmd.Parameters.AddWithValue("@stock", nuevo.Stock);
                    cmd.Parameters.AddWithValue("@imagen", nuevo.ImagenUrl);
                    cmd.ExecuteNonQuery();
                }
            }

            lblEstado.Text = "<div class='alert alert-success'>Producto guardado en el catálogo.</div>";
            txtNombre.Text = "";
            txtDescripcion.Text = "";
            txtPrecio.Text = "";
            txtStock.Text = "";
            txtImagen.Text = "";
        }
    }
}
