2.2. Expresiones Lambda y Manejo de Excepciones

[span_21](start_span)[span_21](end_span)

Las expresiones Lambda permiten implementar programación funcional sintética[span_22](start_span)[span_22](end_span). Para controlar errores en tiempo de ejecución se utiliza el bloque try-catch-finally[span_23](start_span)[span_23](end_span).

import java.util.List;

public class EjemploFuncional {
    public static void main(String[] args) {
        // Expresiones Lambda (forEach)[span_24](start_span)[span_24](end_span)
        List<String> saberes = List.of("Sistemas", "Objetos", "Bases de Datos");
        saberes.forEach(saber -> System.out.println("Saber: " + saber));

        // Control de Excepciones Básicas[span_25](start_span)[span_25](end_span)
        try {
            int resultado = 10 / 0;
        } catch (ArithmeticException e) {
            System.err.println("Error: División entre cero no permitida.");
        } finally {
            System.out.println("Operación finalizada.");[span_26](start_span)[span_26](end_span)
        }
    }
}

import java.sql.Connection;
import java.sql.DriverManager;
import java.sql.PreparedStatement;

public class ConexionBD {
    public void guardarMateria(String codigo, String nombre) {
        String url = "jdbc:mysql://localhost:3306/sistema_db";
        String sql = "INSERT INTO materias (codigo, nombre) VALUES (?, ?)";

        // Try-with-resources (cierre automático de conexión)
        try (Connection conn = DriverManager.getConnection(url, "user", "pass");
             PreparedStatement stmt = conn.prepareStatement(sql)) {
            
            stmt.setString(1, codigo);
            stmt.setString(2, nombre);
            stmt.executeUpdate();[span_40](start_span)[span_40](end_span)
            System.out.println("Registro insertado con éxito.");
        } catch (Exception e) {
            e.printStackTrace();
        }
    }
}
