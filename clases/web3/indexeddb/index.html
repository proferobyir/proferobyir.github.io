¿Qué es IndexedDB y qué es un ObjectStore?

IndexedDB es una base de datos NoSQL transaccional del lado del cliente integrada en el navegador. Permite almacenar grandes volúmenes de datos estructurados de forma asíncrona.

Un ObjectStore actúa de forma similar a una "tabla" en bases de datos relacionales. Es el contenedor donde se organizan los registros (objetos JavaScript) dentro de la base de datos.

Sección 1: Inicialización y Conexión Base

Abre una solicitud a la base de datos y maneja eventos clave como onupgradeneeded.

const request = indexedDB.open("EscuelaDB", 1);
let db;

request.onupgradeneeded = (e) => {
  db = e.target.result;
  if (!db.objectStoreNames.contains("alumnos")) {
    db.createObjectStore("alumnos", { keyPath: "id" });
  }
};

request.onsuccess = (e) => {
  db = e.target.result;
  console.log("1. Conexión exitosa a IndexedDB");
};

Sección 2: Inserción de Datos (Método .add)

Inserta un nuevo registro asegurando que no se sobrescriba uno existente.

function agregarAlumno(alumno) {
  const trans = db.transaction(["alumnos"], "readwrite");
  const store = trans.objectStore("alumnos");
  const req = store.add(alumno); // Falla si el id ya existe

  req.onsuccess = () => console.log("2. Alumno agregado correctamente");
  req.onerror = (e) => console.error("Error al agregar:", e.target.error);
}

Sección 3: Inserción o Actualización (Método .put)

Inserta un registro nuevo o lo actualiza si la clave primaria ya existe.

function guardarOActualizar(alumno) {
  const trans = db.transaction(["alumnos"], "readwrite");
  const store = trans.objectStore("alumnos");
  store.put(alumno); // Inserta o actualiza automáticamente

  trans.oncomplete = () => console.log("3. Operación .put completada con éxito");
}

Sección 4: Lectura de un Registro por Clave (.get)

Recupera un objeto específico usando su clave primaria única.

function obtenerAlumno(id) {
  const trans = db.transaction(["alumnos"], "readonly");
  const store = trans.objectStore("alumnos");
  const req = store.get(id);

  req.onsuccess = (e) => {
    const resultado = e.target.result;
    if (resultado) {
      console.log("4. Alumno encontrado:", resultado);
    } else {
      console.log("4. No existe un registro con ese ID");
    }
  };
}

Sección 5: Listar Todos los Registros (.getAll)

Obtiene una copia completa de todos los elementos almacenados en el ObjectStore.

function listarTodos() {
  const trans = db.transaction(["alumnos"], "readonly");
  const store = trans.objectStore("alumnos");
  const req = store.getAll();

  req.onsuccess = (e) => {
    console.log("5. Listado completo de alumnos:", e.target.result);
  };
}

Sección 6: Eliminación de un Registro (.delete)

Borra un registro individual utilizando su identificador único.

function eliminarAlumno(id) {
  const trans = db.transaction(["alumnos"], "readwrite");
  const store = trans.objectStore("alumnos");
  const req = store.delete(id);

  req.onsuccess = () => console.log(`6. Alumno con ID ${id} eliminado con éxito`);
}

Sección 7: Uso de Cursores para Iterar Registros

Recorre los elementos uno a uno de forma eficiente sin cargar todo en memoria a la vez.

function iterarConCursor() {
  const trans = db.transaction(["alumnos"], "readonly");
  const store = trans.objectStore("alumnos");
  const request = store.openCursor();

  request.onsuccess = (e) => {
    const cursor = e.target.result;
    if (cursor) {
      console.log("7. Cursor leyendo ID:", cursor.key, "Datos:", cursor.value);
      cursor.continue(); // Avanza al siguiente registro
    } else {
      console.log("7. Fin del recorrido con cursor.");
    }
  };
}

Sección 8: Creación y Consulta por Índices

Permite buscar registros mediante propiedades secundarias (por ejemplo, el campo email).

// Nota: El índice se crea en onupgradeneeded:
// store.createIndex("email_idx", "email", { unique: true });

function buscarPorEmail(emailBuscado) {
  const trans = db.transaction(["alumnos"], "readonly");
  const store = trans.objectStore("alumnos");
  const index = store.index("email_idx");
  const req = index.get(emailBuscado);

  req.onsuccess = (e) => {
    console.log("8. Búsqueda por índice de email:", e.target.result);
  };
}

Sección 9: Limpiar Todo el ObjectStore (.clear)

Elimina masivamente todos los registros contenidos dentro del almacén.

function vaciarBaseDeDatos() {
  const trans = db.transaction(["alumnos"], "readwrite");
  const store = trans.objectStore("alumnos");
  const req = store.clear();

  req.onsuccess = () => console.log("9. Todos los registros han sido borrados.");
}

Sección 10: Manejo de Errores y Eventos de Transacción

Monitorea cuando una transacción finaliza por completo o si ocurre algún fallo global.

function transaccionSegura(nuevoAlumno) {
  const trans = db.transaction(["alumnos"], "readwrite");
  const store = trans.objectStore("alumnos");

  store.put(nuevoAlumno);

  trans.oncomplete = () => {
    console.log("10. Transacción completada con éxito en la capa de persistencia.");
  };

  trans.onerror = (e) => {
    console.error("10. Error crítico en la transacción:", e.target.error);
  };
}
