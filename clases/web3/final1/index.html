1. Middleware de Autenticación JWT (Back-End)

Guarda este archivo como middleware/auth.js. Este componente protege los endpoints del servidor validando el token enviado en la cabecera Authorization: Bearer <token>.

middleware/auth.js (Node.js + Express) 
const jwt = require('jsonwebtoken');
const SECRET_KEY = 'clave_secreta_super_segura';

function verificarToken(req, res, next) {
  // Obtener el header 'Authorization'
  const authHeader = req.headers['authorization'];
  // Formato esperado: "Bearer TOKEN"
  const token = authHeader && authHeader.split(' ')[1];

  if (!token) {
    return res.status(401).json({ error: 'Acceso denegado. Token no proporcionado.' });
  }

  jwt.verify(token, SECRET_KEY, (err, usuario) => {
    if (err) {
      return res.status(403).json({ error: 'Token inválido o expirado.' });
    }
    req.usuario = usuario;
    next(); // Continuar hacia la ruta del CRUD
  });
}

module.exports = verificarToken;

3. Lógica del Cliente Fetch API (Front-End JavaScript)

Este archivo (app.js) se encarga de verificar la sesión en el cliente y comunicarse de forma asíncrona mediante fetch enviando las credenciales JWT.

public/app.js (JavaScript Asíncrono + DOM) 
// 1. VERIFICACIÓN DE SESIÓN EN FRONT-END
const token = localStorage.getItem('token');
if (!token) {
  // Si no hay token, redirigir inmediatamente al login
  window.location.href = 'login.html';
}

const API_URL = '/api/estudiantes';
let modalBS = null;

document.addEventListener('DOMContentLoaded', () => {
  modalBS = new bootstrap.Modal(document.getElementById('modalEstudiante'));
  cargarEstudiantes();
});

// Cabecera global con token JWT para peticiones seguras
function getAuthHeaders() {
  return {
    'Content-Type': 'application/json',
    'Authorization': `Bearer ${localStorage.getItem('token')}`
  };
}

// 2. OBTENER Y RENDERIZAR REGISTROS (READ)
async function cargarEstudiantes() {
  try {
    const res = await fetch(API_URL, { headers: getAuthHeaders() });
    if (res.status === 401 || res.status === 403) return cerrarSesion();
    
    const estudiantes = await res.json();
    const tbody = document.getElementById('tablaEstudiantes');
    tbody.innerHTML = '';

    estudiantes.forEach(est => {
      tbody.innerHTML += `
        <tr>
          <td>${est.id}</td>
          <td class="fw-semibold">${est.nombre}</td>
          <td>${est.email}</td>
          <td>${est.carrera}</td>
          <td class="text-end">
            <button class="btn btn-sm btn-outline-primary me-1" onclick="abrirModalEditar(${est.id}, '${est.nombre}', '${est.email}', '${est.carrera}')">✏️ Editar</button>
            <button class="btn btn-sm btn-outline-danger" onclick="eliminarEstudiante(${est.id})">🗑️ Eliminar</button>
          </td>
        </tr>
      `;
    });
  } catch (error) {
    console.error('Error cargando estudiantes:', error);
  }
}

// 3. CREAR Y ACTUALIZAR (CREATE / UPDATE)
function abrirModalCrear() {
  document.getElementById('formEstudiante').reset();
  document.getElementById('estudianteId').value = '';
  document.getElementById('modalTitulo').innerText = 'Nuevo Estudiante';
  modalBS.show();
}

function abrirModalEditar(id, nombre, email, carrera) {
  document.getElementById('estudianteId').value = id;
  document.getElementById('nombre').value = nombre;
  document.getElementById('email').value = email;
  document.getElementById('carrera').value = carrera;
  document.getElementById('modalTitulo').innerText = 'Editar Estudiante';
  modalBS.show();
}

async function guardarEstudiante(e) {
  e.preventDefault();
  const id = document.getElementById('estudianteId').value;
  const datos = {
    nombre: document.getElementById('nombre').value,
    email: document.getElementById('email').value,
    carrera: document.getElementById('carrera').value
  };

  const metodo = id ? 'PUT' : 'POST';
  const url = id ? `${API_URL}/${id}` : API_URL;

  await fetch(url, {
    method: metodo,
    headers: getAuthHeaders(),
    body: JSON.stringify(datos)
  });

  modalBS.hide();
  cargarEstudiantes();
}

// 4. ELIMINAR (DELETE)
async function eliminarEstudiante(id) {
  if (confirm('¿Está seguro de eliminar este registro?')) {
    await fetch(`${API_URL}/${id}`, {
      method: 'DELETE',
      headers: getAuthHeaders()
    });
    cargarEstudiantes();
  }
}

// 5. LOGOUT
function cerrarSesion() {
  localStorage.removeItem('token');
  window.location.href = 'login.html';
}
