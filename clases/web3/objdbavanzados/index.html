1. Fundamentos Teóricos: Transacciones, ACID y Símbolo @

¿Qué es una Transacción?

Es un conjunto de una o varias operaciones SQL (INSERT, UPDATE, DELETE) que se ejecutan como una sola unidad indivisible de trabajo. O se ejecutan todas con éxito, o no se ejecuta ninguna (volviendo al estado original).

Propiedades ACID de la Base de Datos:
  • A - Atomicidad (Atomic): Principio de "todo o nada". Si una sola consulta dentro del bloque falla, se revierte toda la operación.
  • C - Consistencia (Consistency):

    Significa que la base de datos nunca viola sus propias reglas del juego (claves primarias, claves foráneas, restricciones de no vacíos NOT NULL o validaciones CHECK).

    Ejemplo didáctico: Si tienes una regla que dice que el inventario no puede ser menor a cero (stock >= 0) y una transacción intenta vender más de lo existente dejando el inventario en -2, la base de datos aborta la operación inmediatamente. La base de datos prefiere cancelar la transacción antes que guardar datos ilógicos o corruptos.

  • I - Aislamiento (Isolation): Transacciones concurrentes no interfieren entre sí. Los datos intermedios no se ven hasta confirmar.
  • D - Durabilidad (Durability): Una vez confirmada la transacción (COMMIT), los cambios se guardan permanentemente en disco.
Uso del símbolo @ en MariaDB / MySQL:

El símbolo @ antepuesto a una palabra identifica una Variable de Sesión de Usuario (ej. @pedido_id).

  • Variables Locales (sin @): Se declaran con DECLARE y solo existen dentro de un bloque BEGIN ... END de un procedimiento o función.
  • Variables de Sesión (con @): Viven durante toda la conexión del usuario. Se usan para capturar valores generados (como LAST_INSERT_ID()) o para leer parámetros de salida (OUT) en procedimientos almacenados desde la consola o desde Node.js.

2. Script SQL Completo para Pruebas (Tablas, Triggers, Vistas, Funciones y Procedimientos)

Copia y pega este script completo en tu cliente de base de datos (como DBeaver, phpMyAdmin o MySQL Workbench) para crear la estructura base que te permitirá realizar todas las pruebas prácticas de este módulo.

-- ========================================================
-- SCRIPT DE CREACIÓN Y PRUEBAS: OBJETOS AVANZADOS
-- ========================================================

DROP DATABASE IF EXISTS db_tienda_avanzada;
CREATE DATABASE db_tienda_avanzada;
USE db_tienda_avanzada;

-- 1. CREACIÓN DE TABLAS BASE
CREATE TABLE clientes (
    id INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(100) NOT NULL,
    email VARCHAR(100) UNIQUE NOT NULL
);

CREATE TABLE productos (
    id INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(100) NOT NULL,
    precio DECIMAL(10, 2) NOT NULL,
    stock INT NOT NULL
);

CREATE TABLE pedidos (
    id INT AUTO_INCREMENT PRIMARY KEY,
    cliente_id INT NOT NULL,
    fecha DATETIME DEFAULT CURRENT_TIMESTAMP,
    total DECIMAL(10, 2) NOT NULL,
    FOREIGN KEY (cliente_id) REFERENCES clientes(id)
);

CREATE TABLE detalles_pedido (
    id INT AUTO_INCREMENT PRIMARY KEY,
    pedido_id INT NOT NULL,
    producto_id INT NOT NULL,
    cantidad INT NOT NULL,
    precio DECIMAL(10, 2) NOT NULL,
    FOREIGN KEY (pedido_id) REFERENCES pedidos(id),
    FOREIGN KEY (producto_id) REFERENCES productos(id)
);

CREATE TABLE historial_precios (
    id INT AUTO_INCREMENT PRIMARY KEY,
    producto_id INT NOT NULL,
    precio_anterior DECIMAL(10, 2) NOT NULL,
    precio_nuevo DECIMAL(10, 2) NOT NULL,
    fecha_registro DATETIME DEFAULT CURRENT_TIMESTAMP
);

('Carlos Gómez', 'carlos@email.com');

INSERT INTO productos (nombre, precio, stock) VALUES

-- 2. CREACIÓN DE OBJETOS AVANZADOS (DELIMITER)
DELIMITER //

-- Trigger: Auditoría de precios
CREATE TRIGGER trg_auditoria_precio_producto
AFTER UPDATE ON productos
FOR EACH ROW
BEGIN
    IF OLD.precio <> NEW.precio THEN
        INSERT INTO historial_precios (producto_id, precio_anterior, precio_nuevo)
        VALUES (OLD.id, OLD.precio, NEW.precio);
    END IF;
END //

-- Función: Cálculo de IVA
CREATE FUNCTION fn_calcular_iva(monto DECIMAL(10,2), porcentaje DECIMAL(4,2))
RETURNS DECIMAL(10,2)
DETERMINISTIC
BEGIN
    RETURN monto * (porcentaje / 100);
END //

-- Procedimiento Almacenado: Registro seguro de compras con Transacción y Manejo de Errores
CREATE PROCEDURE sp_registrar_compra(
    IN p_cliente_id INT,
    IN p_producto_id INT,
    IN p_cantidad INT,
    OUT p_resultado VARCHAR(100)
)
BEGIN
    DECLARE v_stock_actual INT;
    DECLARE v_precio DECIMAL(10,2);
    
    DECLARE EXIT HANDLER FOR SQLEXCEPTION
    BEGIN
        ROLLBACK;
        SET p_resultado = 'ERROR: Transacción abortada por fallo en el sistema.';
    END;

    START TRANSACTION;

    SELECT stock, precio INTO v_stock_actual, v_precio 
    FROM productos WHERE id = p_producto_id;

    IF v_stock_actual >= p_cantidad THEN
        UPDATE productos SET stock = stock - p_cantidad WHERE id = p_producto_id;
        
        INSERT INTO pedidos (cliente_id, fecha, total) 
        VALUES (p_cliente_id, NOW(), v_precio * p_cantidad);
        
        COMMIT;
        SET p_resultado = 'ÉXITO: Compra registrada correctamente.';
    ELSE
        ROLLBACK;
        SET p_resultado = 'RECHAZADO: Stock insuficiente.';
    END IF;
END //

DELIMITER ;

-- 3. CREACIÓN DE VISTA
CREATE VIEW v_resumen_pedidos AS
SELECT 
    p.id AS pedido_id,
    c.nombre AS cliente,
    c.email AS email_cliente,
    p.total,
    p.fecha
FROM pedidos p
INNER JOIN clientes c ON p.cliente_id = c.id;
Explicación del script integral: Este código inicializa la base de datos completa con tablas relacionales normalizadas, datos de prueba iniciales, un disparador automático de auditoría, una función matemática reutilizable, un procedimiento almacenado robusto con control transaccional ACID y una vista relacional optimizada.

3. Desglose Didáctico y Explicación de Cada Objeto de Base de Datos

A continuación, examinamos paso a paso cómo funciona cada uno de los componentes avanzados que acabamos de crear.

A. Triggers (Disparadores)

Un disparador es un bloque de código SQL que se ejecuta de manera totalmente automática en respuesta a un evento específico (INSERT, UPDATE, DELETE) ocurrido en una tabla.

CREATE TRIGGER trg_auditoria_precio_producto
AFTER UPDATE ON productos
FOR EACH ROW
BEGIN
    IF OLD.precio <> NEW.precio THEN
        INSERT INTO historial_precios (producto_id, precio_anterior, precio_nuevo)
        VALUES (OLD.id, OLD.precio, NEW.precio);
    END IF;
END
Detalles clave:
  • AFTER UPDATE ON productos FOR EACH ROW: Se activa tras modificarse cualquier fila en la tabla productos.
  • OLD y NEW: Permiten comparar el estado anterior y el estado posterior del registro modificado.

B. Transacciones Manuales y Sentencias de Control

Permiten agrupar operaciones críticas asegurando consistencia.

START TRANSACTION;

INSERT INTO pedidos (cliente_id, fecha, total) VALUES (1, NOW(), 150.00);
SET @pedido_id = LAST_INSERT_ID();

INSERT INTO detalles_pedido (pedido_id, producto_id, cantidad, precio) VALUES (@pedido_id, 5, 2, 75.00);
UPDATE productos SET stock = stock - 2 WHERE id = 5;

COMMIT; -- O ROLLBACK en caso de fallo
Detalles clave: START TRANSACTION abre el bloque atómico, LAST_INSERT_ID() rescata el ID autoincremental generado y almacenado temporalmente en una variable de sesión (@), y COMMIT sella los cambios permanentemente.

C. Vistas (`VIEW`)

Una vista es una consulta SQL pre-almacenada que se comporta conceptualmente como una tabla virtual.

CREATE VIEW v_resumen_pedidos AS
SELECT 
    p.id AS pedido_id,
    c.nombre AS cliente,
    c.email AS email_cliente,
    p.total,
    p.fecha
FROM pedidos p
INNER JOIN clientes c ON p.cliente_id = c.id;
Detalles clave: Simplifica drásticamente las consultas desde Node.js al ocultar joins complejos detrás de un simple SELECT * FROM v_resumen_pedidos.

D. Stored Functions (Funciones Almacenadas)

Rutinas personalizadas diseñadas estrictamente para retornar un valor escalar tras realizar cálculos o transformaciones.

CREATE FUNCTION fn_calcular_iva(monto DECIMAL(10,2), porcentaje DECIMAL(4,2))
RETURNS DECIMAL(10,2)
DETERMINISTIC
BEGIN
    RETURN monto * (porcentaje / 100);
END
Detalles clave: Deben especificar explícitamente el tipo de dato devuelto mediante la cláusula RETURNS y pueden ser incrustadas directamente dentro de cualquier instrucción SELECT.

E. Stored Procedures (Procedimientos Almacenados)

Bloques de código complejos capaces de aceptar parámetros de entrada (IN) y salida (OUT), gestionar lógica condicional avanzada, transacciones y control de excepciones (SQLEXCEPTION).

CREATE PROCEDURE sp_registrar_compra(
    IN p_cliente_id INT,
    IN p_producto_id INT,
    IN p_cantidad INT,
    OUT p_resultado VARCHAR(100)
)
BEGIN
    -- Lógica interna de validación de stock, transacciones y control de errores...
END
Detalles clave: Centralizan la lógica de negocio en el servidor de base de datos, reduciendo el tráfico de red y protegiendo la integridad transaccional.

F. Tabla Comparativa: Procedimiento Almacenado vs. Función Almacenada

Para comprender con claridad cuándo utilizar cada objeto dentro del motor de base de datos, revisa la siguiente tabla comparativa:

Característica Procedimiento Almacenado (PROCEDURE) Función Almacenada (FUNCTION)
Valor de retorno No retorna un valor directo; usa parámetros OUT / INOUT o conjuntos de resultados. Debe retornar obligatoriamente un único valor usando la cláusula RETURNS y RETURN.
Forma de invocación Se ejecuta exclusivamente con la sentencia CALL. Se puede invocar dentro de cualquier sentencia SQL (como un SELECT, WHERE o UPDATE).
Parámetros Permite parámetros de tipo IN, OUT e INOUT. Solo permite parámetros de entrada de tipo IN.
Modificación de datos Permite realizar operaciones complejas de manipulación de datos (INSERT, UPDATE, DELETE) y control de transacciones. Está diseñada para cálculos y consultas; por lo general, no se recomienda realizar operaciones de modificación de datos en su interior.
Uso en sentencias SQL No se puede insertar directamente dentro de una consulta SQL normal. Se puede anidar e integrar directamente dentro de otras consultas SQL.
Resumen de selección: Utiliza un Procedimiento Almacenado cuando necesites alterar múltiples tablas, ejecutar transacciones (COMMIT / ROLLBACK) o retornar mensajes de estado. Utiliza una Función Almacenada cuando requieras realizar cálculos matemáticos, formateo de cadenas o validaciones rápidas que necesiten ser incrustadas directamente en el flujo de una consulta SELECT.

4. Integración API REST en Express (Patrón MVC)

Consumo de los Objetos Avanzados creados en MariaDB mediante funciones exportadas dentro del Modelo y Controlador de Express con el paquete mysql2/promise.

1. Modelo de Datos (`models/pedidoModel.js`)

const db = require('../config/db');

// Función exportada para consultar la vista
const obtenerResumenPedidos = async () => {
  const [rows] = await db.query('SELECT * FROM v_resumen_pedidos');
  return rows;
};

// Función exportada para ejecutar el procedimiento almacenado
const registrarCompraSP = async (clienteId, productoId, cantidad) => {
  const [rows] = await db.query(
    'CALL sp_registrar_compra(?, ?, ?, @resultado); SELECT @resultado AS resultado;',
    [clienteId, productoId, cantidad]
  );
  // MariaDB devuelve arreglos múltiples. rows[1][0].resultado extrae la respuesta de @resultado
  return rows[1][0].resultado;
};

module.exports = {
  obtenerResumenPedidos,
  registrarCompraSP
};
Explicación del Modelo:
  • obtenerResumenPedidos: Realiza un SELECT directo sobre la Vista v_resumen_pedidos. Toda la lógica de los JOINs es procesada eficientemente por el motor MariaDB.
  • registrarCompraSP: Ejecuta el procedimiento almacenado con los marcadores de posición ? (evitando inyecciones SQL). Pasa la variable de sesión @resultado para capturar el valor de salida (OUT) y ejecuta inmediatamente el SELECT @resultado en la misma conexión.

2. Controlador API REST (`controllers/pedidoController.js`)

const pedidoModel = require('../models/pedidoModel');

// GET /api/pedidos/resumen
const getResumen = async (req, res) => {
  try {
    const pedidos = await pedidoModel.obtenerResumenPedidos();
    res.status(200).json({
      status: 'success',
      data: pedidos
    });
  } catch (error) {
    res.status(500).json({
      status: 'error',
      message: 'Error al consultar el resumen de pedidos: ' + error.message
    });
  }
};

// POST /api/pedidos/comprar
const crearPedido = async (req, res) => {
  const { clienteId, productoId, cantidad } = req.body;

  if (!clienteId || !productoId || !cantidad) {
    return res.status(400).json({
      status: 'fail',
      message: 'Faltan campos obligatorios en el cuerpo de la petición.'
    });
  }

  try {
    const resultado = await pedidoModel.registrarCompraSP(clienteId, productoId, cantidad);
    
    // Si el procedimiento devolvió un rechazo por stock
    if (resultado.startsWith('RECHAZADO')) {
      return res.status(400).json({ status: 'fail', message: resultado });
    }

    res.status(201).json({
      status: 'success',
      message: resultado
    });
  } catch (error) {
    res.status(500).json({
      status: 'error',
      message: 'Error interno al procesar la compra: ' + error.message
    });
  }
};

module.exports = {
  getResumen,
  crearPedido
};

3. Definición de Rutas (`routes/pedidoRoutes.js`)

const express = require('express');
const router = express.Router();
const pedidoController = require('../controllers/pedidoController');

// Endpoint GET para obtener el resumen mediante la Vista
router.get('/resumen', pedidoController.getResumen);

// Endpoint POST para procesar una compra mediante el Stored Procedure
router.post('/comprar', pedidoController.crearPedido);

module.exports = router;
Explicación del Controlador y Rutas:
  • Mantiene la arquitectura MVC pura basada en funciones, sin clases sintácticas.
  • El Controlador valida los datos de entrada (req.body), invoca la función del Modelo y responde con códigos HTTP estándar (200 OK, 201 Created, 400 Bad Request, 500 Internal Server Error).
  • El archivo de rutas mapea cada verbo HTTP (GET, POST) al método del controlador correspondiente.
