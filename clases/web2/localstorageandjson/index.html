1. Introducción a LocalStorage

Web Storage API proporciona un almacenamiento clave-valor en el navegador. A diferencia de sessionStorage, los datos en localStorage no expiran al cerrar el navegador o apagar el equipo.

Regla fundamental: localStorage solo almacena cadenas de texto (Strings). Todo dato almacenado es convertido implícitamente a texto si no se serializa correctamente.

Métodos Base de la API:

// 1. setItem(clave, valor): Guarda o actualiza un dato tipo String asociado a una clave
localStorage.setItem('clave', 'valor');

// 2. getItem(clave): Retorna el valor en texto guardado con esa clave. Retorna null si no existe
const dato = localStorage.getItem('clave');

// 3. removeItem(clave): Elimina del almacenamiento la clave especificada y su valor
localStorage.removeItem('clave');

// 4. clear(): Borra completamente todos los registros de localStorage asociados al dominio actual
localStorage.clear();

function guardarTextoSimple() {
  const input = document.getElementById('simpleText');

  /*
   * .trim(): Método de las cadenas de texto (String) que remueve los espacios
   * en blanco al inicio y al final del texto. Evita que el usuario guarde
   * una cadena vacía o compuesta unicamente por espacios ("   ").
   */
  const valor = input.value.trim();

  if (valor === "") {
    alert("Por favor escribe un texto válido.");
    return;
  }

  // Guardar en localStorage bajo la clave 'demo_texto_simple'
  localStorage.setItem('demo_texto_simple', valor);

  // Actualizar la interfaz y limpiar el input
  document.getElementById('resultadoSimple').innerText = `Guardado: "${valor}"`;
  input.value = '';
}

function leerTextoSimple() {
  // Lectura del almacenamiento local
  const valorGuardado = localStorage.getItem('demo_texto_simple');

  /*
   * Evaluamos si el valor no es null.
   * getItem() retorna null si la clave consultada no ha sido creada previamente.
   */
  if (valorGuardado !== null) {
    document.getElementById('resultadoSimple').innerText = `Recuperado: "${valorGuardado}"`;
  } else {
    document.getElementById('resultadoSimple').innerText = "No existe la clave 'demo_texto_simple'.";
  }
}

function borrarTextoSimple() {
  // Eliminación explícita de la clave seleccionada
  localStorage.removeItem('demo_texto_simple');
  document.getElementById('resultadoSimple').innerText = "Clave eliminada correctamente.";
}

/*IMPORTANTE SI VAN A USAR EL CÓDIGO EN UN ARCHIVO EXTERNO JS
     Usar defer y DOMContentLoaded al mismo tiempo para envolver un archivo completo es redundante, 
     y además destruye el acceso a las funciones si las llamas con eventos inline (onclick) desde el HTML. 
     Con defer, tu script ya es consciente del DOM.            
*/
let cadenaImagenBase64 = "";

function procesarVistaPrevia(event) {
  /*
   * event.target.files: Arreglo (FileList) con los archivos seleccionados.
   * [0] indica que tomamos el primer archivo elegido por el usuario.
   */
  const archivo = event.target.files[0];
  const imgPreview = document.getElementById('imgPreview');
  const placeholderText = document.getElementById('placeholderText');

  if (archivo) {
    /*
     * FileReader(): Constructor que crea un nuevo objeto lector para inspeccionar
     * archivos en el sistema local de forma asíncrona.
     */
    const lector = new FileReader();

    /*
     * .onload: Manejador de eventos (Event Handler) que se ejecuta cuando
     * la operación de lectura del archivo ha finalizado exitosamente.
     */
    lector.onload = function(e) {
      /*
       * e.target.result: Contiene los datos del archivo expresados como
       * un Data URL (cadena codificada en Base64).
       */
      cadenaImagenBase64 = e.target.result;
      
      // Asignar el Data URL al atributo src de la imagen para vista previa
      imgPreview.src = cadenaImagenBase64;
      imgPreview.style.display = 'block';
      placeholderText.style.display = 'none';
    };

    /*
     * .readAsDataURL(archivo): Método que inicia la lectura del archivo.
     * Una vez finalizado, el atributo result contendrá una URL con los datos
     * del archivo codificados en Base64.
     */
    lector.readAsDataURL(archivo);
  }
}

function ej1_ejecutar() {
  const nombreInput = document.getElementById('ej1_nombre');
  const notaInput = document.getElementById('ej1_nota');

  // 1. Obtener los datos existentes o inicializar un arreglo vacío
  const raw = localStorage.getItem('ej1_estudiantes');

  /*
   * JSON.parse(cadena): Convierte el String guardado a un Arreglo de Objetos real.
   */
  const estudiantes = raw ? JSON.parse(raw) : [];

  /*
   * .push(objeto): Método de Array que añade una nueva estructura al final.
   * parseFloat(): Convierte la cadena numérica recibida a un número decimal flotante.
   * .trim(): Remueve espacios en blanco iniciales y finales.
   */
  estudiantes.push({
    id: Date.now(), // Identificador único por marca de tiempo
    nombre: nombreInput.value.trim(),
    nota: parseFloat(notaInput.value)
  });

  /*
   * JSON.stringify(objeto): Serializa la estructura a String para guardarla en localStorage.
   */
  localStorage.setItem('ej1_estudiantes', JSON.stringify(estudiantes));

  // Renderizar la lista
  document.getElementById('ej1_display').innerText = JSON.stringify(estudiantes, null, 2);
  nombreInput.value = '';
  notaInput.value = '';
}

function ej4_ejecutarAumento() {
  const porcentaje = parseFloat(document.getElementById('ej4_porcentajeInput').value);
  const raw = localStorage.getItem('ej4_precios');
  const lista = raw ? JSON.parse(raw) : [];

  /*
   * Array.prototype.map(callback): Retorna un nuevo arreglo transformando
   * cada uno de los elementos según lo retornado por la función pasada por parámetro.
   * .toFixed(2): Método numérico que redondea el resultado a 2 decimales y lo devuelve como string.
   */
  const listaActualizada = lista.map(producto => {
    return {
      ...producto, // Operador Spread: Copia las propiedades existentes
      precio: Number((producto.precio * (1 + porcentaje / 100)).toFixed(2))
    };
  });

  // Guardar la versión transformada
  localStorage.setItem('ej4_precios', JSON.stringify(listaActualizada));
  document.getElementById('ej4_display').innerText = JSON.stringify(listaActualizada, null, 2);
}

    { id: 2, tarea: "Practicar FileReader" },
    { id: 3, tarea: "Hacer Ejercicios JSON" }

function ej5_ejecutarEliminar() {
  const idAEliminar = document.getElementById('ej5_idEliminar').value;
  const raw = localStorage.getItem('ej5_tareas');
  const tareas = raw ? JSON.parse(raw) : [];

  /*
   * Array.prototype.findIndex(callback): Retorna la posición (índice 0-based) del elemento
   * dentro del arreglo. Devuelve -1 si el elemento no coincide con la condición.
   */
  const indice = tareas.findIndex(item => item.id === Number(idAEliminar));

  if (indice !== -1) {
    /*
     * Array.prototype.splice(posicionInicio, cantidadAEliminar):
     * Modifica el arreglo original eliminando la cantidad de elementos especificada.
     */
    tareas.splice(indice, 1);

    // Sobrescribimos el localStorage con el arreglo modificado
    localStorage.setItem('ej5_tareas', JSON.stringify(tareas));
    document.getElementById('ej5_display').innerText = JSON.stringify(tareas, null, 2);
  } else {
    alert("ID no encontrado");
  }
}

function ej6_ejecutarCalculo() {
  const raw = localStorage.getItem('ej6_carrito');
  const carrito = raw ? JSON.parse(raw) : [];

  /*
   * Array.prototype.reduce(callback, valorInicial):

function ej9_ejecutarAgrupado() {
  const raw = localStorage.getItem('ej9_articulos');
  const articulos = raw ? JSON.parse(raw) : [];

  /*
   * Construimos un objeto dinámico donde las claves serán los nombres de categorías
   * y los valores serán arreglos de artículos pertenecientes a ellas.
   */
  const agrupado = articulos.reduce((acumulador, item) => {
    // Si la categoría aún no existe en el objeto acumulador, la inicializamos como arreglo
    if (!acumulador[item.categoria]) {
      acumulador[item.categoria] = [];
    }
    // Agregamos el item al arreglo de la categoría correspondiente
    acumulador[item.categoria].push(item);
    return acumulador;
  }, {});

  document.getElementById('ej9_display').innerText = JSON.stringify(agrupado, null, 2);
}

function ej10_cargarPorDefecto() {
  localStorage.removeItem('ej10_config');
  const configDefecto = { tema: 'claro', idioma: 'es', notificaciones: true, fuenteTamano: 14 };
  document.getElementById('ej10_display').innerText = JSON.stringify(configDefecto, null, 2);

function ej10_actualizarTemaOscuro() {
  // Configuración predeterminada
  const configDefecto = { tema: 'claro', idioma: 'es', notificaciones: true, fuenteTamano: 14 };

  const raw = localStorage.getItem('ej10_config');
  const configGuardada = raw ? JSON.parse(raw) : {};

  // Nuevas opciones solicitadas por el usuario
  const nuevasOpciones = { tema: 'oscuro', idioma: 'en' };

  /*
   * Object.assign(destino, ...fuentes): Copia todas las propiedades sobrescribiendo
   * las coincidentes de izquierda a derecha. Permite fusionar estados.
   */
  const configFinal = Object.assign({}, configDefecto, configGuardada, nuevasOpciones);

  // Guardar configuración consolidada
  localStorage.setItem('ej10_config', JSON.stringify(configFinal));
  document.getElementById('ej10_display').innerText = JSON.stringify(configFinal, null, 2);
}
