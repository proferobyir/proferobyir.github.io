Guía Didáctica: API REST Segura en C# 5 & MySQL

Aprende a construir desde cero una API REST en Visual Studio 2012 (.NET Framework 4.5) conectada a MySQL. Incluye soporte abierto para CORS, Autenticación con Token, Middleware de Autorización y transacciones de ventas.

1. Configuración Paso a Paso en Visual Studio 2012

Sigue estas instrucciones para crear e integrar la solución en tu entorno local:

  1. Crear el Proyecto: Abre Visual Studio 2012, selecciona File -> New -> Project. Elige ASP.NET MVC 4 Web Application, nómbralo FeriaIutepiAPI y selecciona la plantilla Web API con .NET Framework 4.5.
  2. Agregar Conector MySQL: Haz clic derecho en References -> Add Reference... y busca la librería MySql.Data.dll (Connector/Net de MySQL para .NET 4.5).
    Nota de Descarga: Puedes descargar directamente el conector oficial de MySQL (MySql.Data.dll) necesario para este laboratorio desde el siguiente enlace oficial en Google Drive: 📥 Descargar MySql.Data.dll (Google Drive)
  3. Configurar Web.config: Reemplaza la sección de cadenas de conexión y habilita los encabezados de CORS.
  4. Crear Estructura de Carpetas: En el Explorador de Soluciones, asegúrate de tener las carpetas: Models, Controllers y crea la carpeta Filters.
  5. Agregar los Archivos del Código: Copia y pega los bloques de código que se muestran a continuación en sus respectivas ubicaciones.

2. Cadena de Conexión & CORS Abierto

Ubicación: Web.config

Configura la conexión a la base de datos MySQL feria_iutepi y habilita CORS (Cross-Origin Resource Sharing) para que la API acepte peticiones desde aplicaciones móviles o clientes web.

XML / Web.config 
<?xml version="1.0" encoding="utf-8"?>
<configuration>
  <connectionStrings>
    <!-- Conexión a la base de datos MySQL feria_iutepi -->
    <add name="ConexionMySQL" 
         connectionString="Server=127.0.0.1;Port=3306;Database=feria_iutepi;Uid=root;Pwd=;SslMode=None;" 
         providerName="MySql.Data.MySqlClient" />
  </connectionStrings>

  <system.web>
    <compilation debug="true" targetFramework="4.5" />
    <httpRuntime targetFramework="4.5" />
  </system.web>

  <system.webServer>
    <!-- Configuración de Encabezados CORS para Conexiones Externas -->
    <httpProtocol>
      <customHeaders>
        <add name="Access-Control-Allow-Origin" value="*" />
        <add name="Access-Control-Allow-Methods" value="GET, POST, PUT, DELETE, OPTIONS" />
        <add name="Access-Control-Allow-Headers" value="Content-Type, Accept, Authorization, X-Requested-With" />
      </customHeaders>
    </httpProtocol>

    <handlers>
      <remove name="ExtensionlessUrlHandler-Integrated-4.0" />
      <remove name="OPTIONSVerbHandler" />
      <remove name="TRACEVerbHandler" />
      <add name="ExtensionlessUrlHandler-Integrated-4.0" path="*." verb="*" type="System.Web.Handlers.TransferRequestHandler" preCondition="integratedMode,runtimeVersionv4.0" />
    </handlers>
  </system.webServer>
</configuration>

3. Rutas y Formateador JSON

Ubicación: App_Start/WebApiConfig.cs

Registra el mapa de rutas e impone formato JSON en todas las respuestas HTTP.

C# (.NET 4.5) 
using System;
using System.Net.Http.Headers;
using System.Web.Http;

namespace FeriaIutepiAPI
{
    public static class WebApiConfig
    {
        public static void Register(HttpConfiguration config)
        {
            // Habilitar atributos de rutas
            config.MapHttpAttributeRoutes();

            // Ruta predeterminada
            config.Routes.MapHttpRoute(
                name: "DefaultApi",
                routeTemplate: "api/{controller}/{id}",
                defaults: new { id = RouteParameter.Optional }
            );

            // Forzar respuesta siempre en JSON
            config.Formatters.JsonFormatter.SupportedMediaTypes.Add(new MediaTypeHeaderValue("text/html"));
            config.Formatters.Remove(config.Formatters.XmlFormatter);
        }
    }
}

5. Controlador de Autenticación & Tokens

Ubicación: Controllers/AuthController.cs

Valida usuarios contra la tabla usuarios (contraseñas SHA256) y emite un token de acceso temporal.

C# (.NET 4.5) 
using System;
using System.Collections.Generic;
using System.Configuration;
using System.Net;
using System.Net.Http;
using System.Security.Cryptography;
using System.Text;
using System.Web.Http;
using MySql.Data.MySqlClient;
using FeriaIutepiAPI.Models;

namespace FeriaIutepiAPI.Controllers
{
    [RoutePrefix("api/auth")]
    public class AuthController : ApiController
    {
        private readonly string cadenaConexion = ConfigurationManager.ConnectionStrings["ConexionMySQL"].ConnectionString;

        // Diccionario estático en memoria para almacenar tokens activos
        public static Dictionary<string, string> SesionesActivas = new Dictionary<string, string>();

        private string EncriptarSHA256(string texto)
        {
            using (var sha256 = SHA256.Create())
            {
                byte[] bytes = sha256.ComputeHash(Encoding.UTF8.GetBytes(texto));
                var sb = new StringBuilder();
                foreach (var b in bytes) sb.Append(b.ToString("x2"));
                return sb.ToString();
            }
        }

        // POST: api/auth/login
        [HttpPost]
        [Route("login")]
        public HttpResponseMessage Login([FromBody] LoginRequest credenciales)
        {
            if (credenciales == null || string.IsNullOrEmpty(credenciales.Usuario) || string.IsNullOrEmpty(credenciales.Password))
            {
                return Request.CreateErrorResponse(HttpStatusCode.BadRequest, "Proporcione usuario y contraseña.");
            }

            string passwordHash = EncriptarSHA256(credenciales.Password);
            string rol = string.Empty;
            string nombreCompleto = string.Empty;
            bool activo = false;

            using (var conexion = new MySqlConnection(cadenaConexion))
            {
                string query = "SELECT rol, nombre_completo, estado FROM usuarios WHERE usuario = @usuario AND password = @password";
                using (var cmd = new MySqlCommand(query, conexion))
                {
                    cmd.Parameters.AddWithValue("@usuario", credenciales.Usuario);
                    cmd.Parameters.AddWithValue("@password", passwordHash);

                    conexion.Open();
                    using (var reader = cmd.ExecuteReader())
                    {
                        if (reader.Read())
                        {
                            rol = reader.GetString("rol");
                            nombreCompleto = reader.GetString("nombre_completo");
                            activo = reader.GetBoolean("estado");
                        }
                    }
                }
            }

            if (string.IsNullOrEmpty(rol))
            {
                return Request.CreateErrorResponse(HttpStatusCode.Unauthorized, "Credenciales incorrectas.");
            }

            if (!activo)
            {
                return Request.CreateErrorResponse(HttpStatusCode.Forbidden, "El usuario está inactivo.");
            }

            string token = Guid.NewGuid().ToString();

            lock (SesionesActivas)
            {
                SesionesActivas[token] = credenciales.Usuario;
            }

            return Request.CreateResponse(HttpStatusCode.OK, new TokenResponse
            {
                Token = token,
                Rol = rol,
                NombreCompleto = nombreCompleto
            });
        }
    }
}

6. Middleware Filtro de Autorización

Ubicación: Filters/CustomAuthorizeAttribute.cs

Intercepta las peticiones y bloquea el acceso con HTTP 401 si no se envía un encabezado Authorization: Bearer <token> válido.

C# (.NET 4.5) 
using System.Net;
using System.Net.Http;
using System.Web.Http.Controllers;
using System.Web.Http.Filters;
using FeriaIutepiAPI.Controllers;

namespace FeriaIutepiAPI.Filters
{
    public class CustomAuthorizeAttribute : AuthorizationFilterAttribute
    {
        public override void OnAuthorization(HttpActionContext actionContext)
        {
            var authHeader = actionContext.Request.Headers.Authorization;

            if (authHeader != null && authHeader.Scheme == "Bearer")
            {
                string token = authHeader.Parameter;
                bool tokenValido = false;

                lock (AuthController.SesionesActivas)
                {
                    if (!string.IsNullOrEmpty(token) && AuthController.SesionesActivas.ContainsKey(token))
                    {
                        tokenValido = true;
                    }
                }

                if (tokenValido)
                {
                    base.OnAuthorization(actionContext);
                    return;
                }
            }

            actionContext.Response = actionContext.Request.CreateErrorResponse(
                HttpStatusCode.Unauthorized, 
                "Acceso denegado. Se requiere un token válido en el encabezado Authorization."
            );
        }
    }
}

7. Controlador Protegido de Productos

Ubicación: Controllers/ProductosController.cs

Protegido con el atributo [CustomAuthorize]. Gestiona operaciones CRUD en MySQL.

C# (.NET 4.5) 
using System;
using System.Collections.Generic;
using System.Configuration;
using System.Net;
using System.Net.Http;
using System.Web.Http;
using MySql.Data.MySqlClient;
using FeriaIutepiAPI.Filters;
using FeriaIutepiAPI.Models;

namespace FeriaIutepiAPI.Controllers
{
    [CustomAuthorize] // Middleware aplicado a todo el controlador
    public class ProductosController : ApiController
    {
        private readonly string cadenaConexion = ConfigurationManager.ConnectionStrings["ConexionMySQL"].ConnectionString;

        // GET: api/productos
        [HttpGet]
        public HttpResponseMessage ObtenerTodos()
        {
            try
            {
                var lista = new List<Producto>();
                using (var conexion = new MySqlConnection(cadenaConexion))
                {
                    string sql = @"SELECT p.id, p.categoria_id, c.nombre AS categoria_nombre, p.nombre, p.precio, p.stock 
                                   FROM productos p 
                                   LEFT JOIN categorias c ON p.categoria_id = c.id";

                    using (var cmd = new MySqlCommand(sql, conexion))
                    {
                        conexion.Open();
                        using (var reader = cmd.ExecuteReader())
                        {
                            while (reader.Read())
                            {
                                lista.Add(new Producto
                                {
                                    Id = reader.GetInt32("id"),
                                    CategoriaId = reader.IsDBNull(reader.GetOrdinal("categoria_id")) ? (int?)null : reader.GetInt32("categoria_id"),
                                    CategoriaNombre = reader.IsDBNull(reader.GetOrdinal("categoria_nombre")) ? "Sin Categoría" : reader.GetString("categoria_nombre"),
                                    Nombre = reader.GetString("nombre"),
                                    Precio = reader.GetDecimal("precio"),
                                    Stock = reader.GetInt32("stock")
                                });
                            }
                        }
                    }
                }
                return Request.CreateResponse(HttpStatusCode.OK, lista);
            }
            catch (Exception ex)
            {
                return Request.CreateErrorResponse(HttpStatusCode.InternalServerError, ex.Message);
            }
        }
    }
}

8. Controlador Transaccional de Ventas

Ubicación: Controllers/VentasController.cs

Proceso atómico con MySqlTransaction para insertar en ventas, descontar stock y registrar detalle_ventas.

C# (.NET 4.5 / ACID) 
using System;
using System.Collections.Generic;
using System.Configuration;
using System.Net;
using System.Net.Http;
using System.Web.Http;
using MySql.Data.MySqlClient;
using FeriaIutepiAPI.Filters;
using FeriaIutepiAPI.Models;

namespace FeriaIutepiAPI.Controllers
{
    [CustomAuthorize]
    public class VentasController : ApiController
    {
        private readonly string cadenaConexion = ConfigurationManager.ConnectionStrings["ConexionMySQL"].ConnectionString;

        // POST: api/ventas/registrar
        [HttpPost]
        [Route("api/ventas/registrar")]
        public HttpResponseMessage RegistrarVenta([FromBody] RegistrarVentaDTO dto)
        {
            if (dto == null || dto.Detalles == null || dto.Detalles.Count == 0)
                return Request.CreateErrorResponse(HttpStatusCode.BadRequest, "Detalles de venta incompletos.");

            using (var conexion = new MySqlConnection(cadenaConexion))
            {
                conexion.Open();
                MySqlTransaction transaccion = conexion.BeginTransaction();

                try
                {
                    decimal totalVenta = 0;
                    var listaProcesar = new List<DetalleVenta>();

                    foreach (var item in dto.Detalles)
                    {
                        string sqlProd = "SELECT nombre, precio, stock FROM productos WHERE id = @prodId FOR UPDATE";
                        using (var cmdProd = new MySqlCommand(sqlProd, conexion, transaccion))
                        {
                            cmdProd.Parameters.AddWithValue("@prodId", item.ProductoId);
                            using (var reader = cmdProd.ExecuteReader())
                            {
                                if (!reader.Read())
                                {
                                    reader.Close();
                                    transaccion.Rollback();
                                    return Request.CreateErrorResponse(HttpStatusCode.NotFound, "Producto no encontrado.");
                                }

                                string nombre = reader.GetString("nombre");
                                decimal precio = reader.GetDecimal("precio");
                                int stock = reader.GetInt32("stock");
                                reader.Close();

                                if (stock < item.Cantidad)
                                {
                                    transaccion.Rollback();
                                    return Request.CreateErrorResponse(HttpStatusCode.BadRequest, "Stock insuficiente para " + nombre);
                                }

                                decimal subtotal = precio * item.Cantidad;
                                totalVenta += subtotal;

                                listaProcesar.Add(new DetalleVenta
                                {
                                    ProductoId = item.ProductoId,
                                    Cantidad = item.Cantidad,
                                    Subtotal = subtotal
                                });
                            }
                        }
                    }

                    // Insertar Venta
                    string sqlVenta = "INSERT INTO ventas (cliente_id, fecha, total) VALUES (@cli, NOW(), @tot); SELECT LAST_INSERT_ID();";
                    int ventaId = 0;
                    using (var cmdVenta = new MySqlCommand(sqlVenta, conexion, transaccion))
                    {
                        cmdVenta.Parameters.AddWithValue("@cli", (object)dto.ClienteId ?? DBNull.Value);
                        cmdVenta.Parameters.AddWithValue("@tot", totalVenta);
                        ventaId = Convert.ToInt32(cmdVenta.ExecuteScalar());
                    }

                    // Detalle y Descuento de Stock
                    foreach (var det in listaProcesar)
                    {
                        string sqlDet = "INSERT INTO detalle_ventas (venta_id, producto_id, cantidad, subtotal) VALUES (@vId, @pId, @cant, @sub)";
                        using (var cmdDet = new MySqlCommand(sqlDet, conexion, transaccion))
                        {
                            cmdDet.Parameters.AddWithValue("@vId", ventaId);
                            cmdDet.Parameters.AddWithValue("@pId", det.ProductoId);
                            cmdDet.Parameters.AddWithValue("@cant", det.Cantidad);
                            cmdDet.Parameters.AddWithValue("@sub", det.Subtotal);
                            cmdDet.ExecuteNonQuery();
                        }

                        string sqlStock = "UPDATE productos SET stock = stock - @cant WHERE id = @pId";
                        using (var cmdStock = new MySqlCommand(sqlStock, conexion, transaccion))
                        {
                            cmdStock.Parameters.AddWithValue("@cant", det.Cantidad);
                            cmdStock.Parameters.AddWithValue("@pId", det.ProductoId);
                            cmdStock.ExecuteNonQuery();
                        }
                    }

                    transaccion.Commit();
                    return Request.CreateResponse(HttpStatusCode.Created, new { Mensaje = "Venta exitosa", VentaId = ventaId, Total = totalVenta });
                }
                catch (Exception ex)
                {
                    transaccion.Rollback();
                    return Request.CreateErrorResponse(HttpStatusCode.InternalServerError, ex.Message);
                }
            }
        }
    }
