Guía Práctica: API REST en C# 5 y MySQL

Diseñado para Visual Studio 2012 (.NET Framework 4.5) de forma simple y didáctica

Recurso esencial: Para conectar tu proyecto con MySQL de forma clásica en este entorno offline, puedes descargar el conector oficial MySql.Data.dll desde este enlace de Google Drive y agregarlo a tus referencias.

Paso 1: Configurar CORS y Conexión (Web.config)

Agrega la cadena de conexión a tu base de datos feria_iutepi y habilita los encabezados CORS en el archivo Web.config para permitir peticiones externas de manera limpia.

<configuration>
  <connectionStrings>
    <add name="ConexionMySQL" 
         connectionString="Server=127.0.0.1;Database=feria_iutepi;Uid=root;Pwd=;" 
         providerName="MySql.Data.MySqlClient" />
  </connectionStrings>

  <system.webServer>
    <httpProtocol>
      <customHeaders>
        <!-- Permite conexiones de origen cruzado (CORS) -->
        <add name="Access-Control-Allow-Origin" value="*" />
        <add name="Access-Control-Allow-Methods" value="GET, POST, PUT, DELETE, OPTIONS" />
        <add name="Access-Control-Allow-Headers" value="Content-Type, Authorization" />
      <!-- /customHeaders -->
    </httpProtocol>
  </system.webServer>
</configuration>

Paso 2: Filtro de Seguridad / Middleware (CustomAuthorizeAttribute.cs)

Este filtro intercepta cada solicitud HTTP para verificar que el cliente envíe un Token válido en el encabezado (Authorization: Bearer <token>).

using System.Net;
using System.Net.Http;
using System.Web.Http.Controllers;
using System.Web.Http.Filters;
using FeriaIutepiAPI.Controllers;

namespace FeriaIutepiAPI.Filters
{
    public class CustomAuthorizeAttribute : AuthorizationFilterAttribute
    {
        public override void OnAuthorization(HttpActionContext actionContext)
        {
            var authHeader = actionContext.Request.Headers.Authorization;

            // Valida si se incluyó el token bajo el esquema Bearer
            if (authHeader != null && authHeader.Scheme == "Bearer")
            {
                string token = authHeader.Parameter;
                bool tokenValido = false;

                lock (AuthController.SesionesActivas)
                {
                    if (!string.IsNullOrEmpty(token) && AuthController.SesionesActivas.ContainsKey(token))
                    {
                        tokenValido = true;
                    }
                }

                if (tokenValido)
                {
                    base.OnAuthorization(actionContext);
                    return; // Permite el acceso
                }
            }

            // Rechaza la solicitud si no hay token o es inválido (Error 401)
            actionContext.Response = actionContext.Request.CreateErrorResponse(
                HttpStatusCode.Unauthorized, 
                "Acceso denegado. Debe iniciar sesión para obtener un token."
            );
        }
    }
}

Paso 3: Controlador de Autenticación (AuthController.cs)

Valida las credenciales contra la tabla usuarios usando el cifrado SHA256 nativo de la base de datos y emite un token GUID temporal.

using System;
using System.Collections.Generic;
using System.Configuration;
using System.Data;
using System.Net;
using System.Net.Http;
using System.Security.Cryptography;
using System.Text;
using System.Web.Http;
using MySql.Data.MySqlClient;

namespace FeriaIutepiAPI.Controllers
{
    [RoutePrefix("api/auth")]
    public class AuthController : ApiController
    {
        private readonly string conexionDB = ConfigurationManager.ConnectionStrings["ConexionMySQL"].ConnectionString;
        
        // Memoria estática para almacenar sesiones activas (Token -> Usuario)
        public static Dictionary<string, string> SesionesActivas = new Dictionary<string, string>();

        // Función auxiliar para cifrar texto en SHA256
        private string EncriptarSHA256(string texto)
        {
            using (var sha256 = SHA256.Create())
            {
                byte[] bytes = sha256.ComputeHash(Encoding.UTF8.GetBytes(texto));
                var sb = new StringBuilder();
                foreach (var b in bytes) sb.Append(b.ToString("x2"));
                return sb.ToString();
            }
        }

        // POST: api/auth/login
        [HttpPost]
        [Route("login")]
        public HttpResponseMessage Login([FromBody] dynamic credenciales)
        {
            string usuario = credenciales?.usuario;
            string password = credenciales?.password;

            if (string.IsNullOrEmpty(usuario) || string.IsNullOrEmpty(password))
                return Request.CreateErrorResponse(HttpStatusCode.BadRequest, "Usuario y contraseña requeridos.");

            string passHash = EncriptarSHA256(password);
            string rol = string.Empty;

            using (var conexion = new MySqlConnection(conexionDB))
            {
                string query = "SELECT rol FROM usuarios WHERE usuario = @usu AND password = @pass AND estado = 1";
                using (var cmd = new MySqlCommand(query, conexion))
                {
                    cmd.Parameters.AddWithValue("@usu", usuario);
                    cmd.Parameters.AddWithValue("@pass", passHash);
                    conexion.Open();
                    var resultado = cmd.ExecuteScalar();
                    if (resultado != null) rol = resultado.ToString();
                }
            }

            if (string.IsNullOrEmpty(rol))
                return Request.CreateErrorResponse(HttpStatusCode.Unauthorized, "Credenciales incorrectas o usuario inactivo.");

            // Generar Token único
            string token = Guid.NewGuid().ToString();
            lock (SesionesActivas) { SesionesActivas[token] = usuario; }

            return Request.CreateResponse(HttpStatusCode.OK, new { Token = token, Rol = rol });
        }
    }
}

Paso 4: Controlador Protegido (ProductosController.cs)

Utiliza ADO.NET clásico y está protegido automáticamente por el middleware [CustomAuthorize].

using System.Collections.Generic;
using System.Configuration;
using System.Net;
using System.Net.Http;
using System.Web.Http;
using MySql.Data.MySqlClient;
using FeriaIutepiAPI.Filters;

namespace FeriaIutepiAPI.Controllers
{
    [CustomAuthorize] // <--- ¡Protege todo el controlador con el token de seguridad!
    public class ProductosController : ApiController
    {
        private readonly string conexionDB = ConfigurationManager.ConnectionStrings["ConexionMySQL"].ConnectionString;

        // GET: api/productos
        [HttpGet]
        public HttpResponseMessage Get()
        {
            var lista = new List<object>();

            using (var conexion = new MySqlConnection(conexionDB))
            {
                string query = "SELECT id, nombre, precio, stock FROM productos";
                using (var cmd = new MySqlCommand(query, conexion))
                {
                    conexion.Open();
                    using (var reader = cmd.ExecuteReader())
                    {
                        while (reader.Read())
                        {
                            lista.Add(new {
                                Id = reader.GetInt32("id"),
                                Nombre = reader.GetString("nombre"),
                                Precio = reader.GetDecimal("precio"),
                                Stock = reader.GetInt32("stock")
                            });
                        }
                    }
                }
            }
            return Request.CreateResponse(HttpStatusCode.OK, lista);
        }
    }
}
