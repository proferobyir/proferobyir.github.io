(Controllers/AuthController.cs)</h2> <p>Crea este controlador en la carpeta <code>Controllers</code>. Valida las credenciales contra la tabla <code>usuarios</code> y emite un token temporal guardado en una lista estática. </p> <div class="code-box"> <pre> using System; using System.Collections.Generic; using System.Configuration; using System.Net; using System.Net.Http; using System.Security.Cryptography; // Necesario para SHA256 using System.Text; // Necesario para Encoding.UTF8 using System.Web.Http; using MySql.Data.MySqlClient; using APIFeria.Models; namespace APIFeria.Controllers { public class AuthController : ApiController { private readonly string conexionDB = ConfigurationManager.ConnectionStrings["ConexionMySQL"].ConnectionString; // Memoria estática compartida para almacenar sesiones activas (Token -> Usuario) public static Dictionary<string, string> Sesiones = new Dictionary<string, string>(); // Método privado auxiliar para cifrar el texto en SHA256 private string EncriptarSHA256(string texto) { using (var sha256 = SHA256.Create()) { byte[] bytes = sha256.ComputeHash(Encoding.UTF8.GetBytes(texto)); var sb = new StringBuilder(); foreach (var b in bytes) sb.Append(b.ToString("x2")); return sb.ToString(); } } // POST: api/Auth/Login [HttpPost] public HttpResponseMessage Login([FromBody] LoginModel credenciales) { if (credenciales == null || string.IsNullOrEmpty(credenciales.usuario) || string.IsNullOrEmpty(credenciales.password)) return Request.CreateErrorResponse(HttpStatusCode.BadRequest, "Debe enviar usuario y contraseña."); // Ciframos lo que escribe el alumno para compararlo con el hash de la base de datos string passHash = EncriptarSHA256(credenciales.password); bool usuarioValido = false; using (var conexion = new MySqlConnection(conexionDB)) { string query = "SELECT COUNT(*) FROM usuarios WHERE usuario = @usu AND password = @pass AND estado = 1"; using (var cmd = new MySqlCommand(query, conexion)) { cmd.Parameters.AddWithValue("@usu", credenciales.usuario); cmd.Parameters.AddWithValue("@pass", passHash); // Comparamos contra el hash SHA256 conexion.Open(); int resultado = Convert.ToInt32(cmd.ExecuteScalar()); if (resultado > 0) usuarioValido = true; } } if (!usuarioValido) return Request.CreateErrorResponse(HttpStatusCode.Unauthorized, "Credenciales incorrectas."); // Generar Token único string token = Guid.NewGuid().ToString(); lock (Sesiones) { Sesiones[token] = credenciales.usuario; } return Request.CreateResponse(HttpStatusCode.OK, new { Token = token, Mensaje = "¡Login exitoso!" }); } } } </pre> </div> </section> <!-- PASO 4 --> <section> <h2>Paso 4: Controlador Protegido con Método Privado (Controllers/ProductosController.cs)</h2> <p>Crea este controlador en la carpeta <code>Controllers</code>. Aquí aplicamos el concepto de <strong>método privado</strong> para encapsular la validación del Token antes de entregar los datos de la base de datos. </p> <div class="code-box"> <pre>using System.Collections.Generic; using System.Configuration; using System.Linq; using System.Net; using System.Net.Http; using System.Web.Http; using MySql.Data.MySqlClient; namespace APIFeria.Controllers { public class ProductosController : ApiController { private readonly string conexionDB = ConfigurationManager.ConnectionStrings["ConexionMySQL"].ConnectionString; // ========================================================================= // MÉTODO PRIVADO DE SEGURIDAD (Concepto clave de visibilidad) // Es 'private' porque solo este controlador necesita ejecutar esta validación // internamente y no debe ser expuesto como una ruta web pública para clientes. // ========================================================================= private bool ValidarToken() { // Verificamos si en los headers (cabeceras HTTP) de la petición viene el campo "Token" if (Request.Headers.Contains("Token")) { string tokenEnviado = Request.Headers.GetValues("Token").FirstOrDefault(); // Comprobamos si el token existe dentro de las sesiones activas del AuthController if (!string.IsNullOrEmpty(tokenEnviado) && AuthController.Sesiones.ContainsKey(tokenEnviado)) { return true; // ¡Token legítimo y válido! } } return false; // ¡Token ausente o incorrecto! } // GET: api/Productos (Método público protegido) [HttpGet] public HttpResponseMessage Get() { // Llamamos al método privado de seguridad antes de consultar la base de datos if (!ValidarToken()) { return Request.CreateErrorResponse(HttpStatusCode.Unauthorized, "Acceso denegado. Falta el Token o es inválido."); } var lista = new List<object>(); // Realizamos la consulta a la tabla productos con ADO.NET using (var conexion = new MySqlConnection(conexionDB)) { string query = "SELECT id, nombre, precio, stock FROM productos"; using (var cmd = new MySqlCommand(query, conexion)) { conexion.Open(); using (var reader = cmd.ExecuteReader()) { // Recorremos fila por fila los registros obtenidos de MySQL while (reader.Read()) { lista.Add(new { Id = reader.GetInt32("id"), Nombre = reader.GetString("nombre"), Precio = reader.GetDecimal("precio"), Stock = reader.GetInt32("stock") }); } } } } // Web API convierte automáticamente la lista de objetos anónimos a formato JSON return Request.CreateResponse(HttpStatusCode.OK, lista); } } }</pre> </div> </section> <footer> <p>Guía de Laboratorio Académico - C# 5 / .NET 4.5 / MySQL - Profesor: Ing.
